# ============================================================
# 1. EXECUTAR NO POWERSHELL DO TEU PC

Test-NetConnection takeacab.online -Port 443
Test-NetConnection 34.62.101.141 -Port 80
Test-NetConnection 34.62.101.141 -Port 443

# https -> HTTP/1.1 200 OK
# ============================================================

curl.exe -I http://takeacab.online
curl.exe -I https://takeacab.online

# ============================================================
# 6. EXECUTAR NO SSH DO lb1
# LB1 -> Web Servers :8080
# Resultado esperado:
# HTTP/1.1 200 OK
# ============================================================

for ip in 10.132.0.3 10.132.0.37 10.132.0.36 10.132.0.38 10.132.0.39; do
  echo "LB1 -> WEB $ip:8080 deve FUNCIONAR"
  curl -I --connect-timeout 5 http://$ip:8080 || echo "FALHOU - VERIFICAR FIREWALL"
done


# ============================================================
# 7. EXECUTAR NO SSH DO lb2
# LB2 -> Web Servers :8080
# Resultado esperado:

# ============================================================

for ip in 10.132.0.3 10.132.0.37 10.132.0.36 10.132.0.38 10.132.0.39; do
  echo "LB2 -> WEB $ip:8080 deve FUNCIONAR"
  curl -I --connect-timeout 5 http://$ip:8080 || echo "FALHOU - VERIFICAR FIREWALL"
done


# ============================================================
# 8. EXECUTAR NO SSH DO as1
# AS1 -> MongoDB :27017
# Como MongoDB nao responde HTTP, usamos curl telnet mode.
# Resultado esperado:
# Connected to 10.132.0.x
# ou erro do tipo "Empty reply", mas depois de conectar.
# Se aparecer "Failed to connect", FALHOU.
# ============================================================

for ip in 10.132.0.4 10.132.0.16 10.132.0.17; do
  echo "AS1 -> DB $ip:27017 deve FUNCIONAR"
  curl -v --connect-timeout 5 telnet://$ip:27017
done


# ============================================================
# 9. EXECUTAR NO SSH DO as2, as3, as4 e as5
# Cada Web Server -> MongoDB :27017
# Resultado esperado:
# Connected to 10.132.0.x
# Se aparecer "Failed to connect", FALHOU.
# ============================================================

for ip in 10.132.0.4 10.132.0.16 10.132.0.17; do
  echo "$(hostname) -> DB $ip:27017 deve FUNCIONAR"
  curl -v --connect-timeout 5 telnet://$ip:27017
done


# ============================================================
# 10. EXECUTAR NO SSH DO lb1
# LB1 -> MongoDB :27017
# Resultado esperado ideal:
# Failed to connect / Connection timed out
# Se aparecer "Connected to", esta ABERTO e deve verificar firewall.
# ============================================================

for ip in 10.132.0.4 10.132.0.16 10.132.0.17; do
  echo "LB1 -> DB $ip:27017 deve FALHAR"
  curl -v --connect-timeout 5 telnet://$ip:27017
done


# ============================================================
# 11. EXECUTAR NO SSH DO lb2
# LB2 -> MongoDB :27017
# Resultado esperado ideal:
# Failed to connect / Connection timed out
# Se aparecer "Connected to", esta ABERTO e deve verificar firewall.
# ============================================================

for ip in 10.132.0.4 10.132.0.16 10.132.0.17; do
  echo "LB2 -> DB $ip:27017 deve FALHAR"
  curl -v --connect-timeout 5 telnet://$ip:27017
done


# ============================================================
# 12. EXECUTAR NO SSH DO db1
# DB1 -> Web Servers :8080
# Resultado esperado ideal:
# Failed to connect / Connection timed out
# Se aparecer HTTP/1.1 200 OK, esta ABERTO e deve verificar firewall.
# ============================================================

for ip in 10.132.0.3 10.132.0.37 10.132.0.36 10.132.0.38 10.132.0.39; do
  echo "DB1 -> WEB $ip:8080 deve FALHAR"
  curl -I --connect-timeout 5 http://$ip:8080 || echo "FALHOU - OK"

# ============================================================
# 13. EXECUTAR NO SSH DO db1
# DB1 -> Load Balancers :80 e :443
# Resultado esperado ideal:
# Failed to connect / Connection timed out
# Se aparecer HTTP/1.1 200 ou 301, esta ABERTO e deve verificar firewall.
# ============================================================

for ip in 10.132.0.2 10.132.0.6; do
  echo "DB1 -> LB $ip:80 deve FALHAR"
  curl -I --connect-timeout 5 http://$ip || echo "FALHOU - OK"

  echo "DB1 -> LB $ip:443 deve FALHAR"
  curl -I -k --connect-timeout 5 https://$ip || echo "FALHOU - OK"
done


# ============================================================
# 14. EXECUTAR NO SSH DO as1
# AS1 -> outros Web Servers :8080
# Resultado esperado ideal:
# Failed to connect / Connection timed out
# Se aparecer HTTP/1.1 200 OK, os Web Servers comunicam entre si.
# ============================================================

for ip in 10.132.0.37 10.132.0.36 10.132.0.38 10.132.0.39; do
  echo "AS1 -> outro WEB $ip:8080 deve FALHAR se nao for necessario"
  curl -I --connect-timeout 5 http://$ip:8080 || echo "FALHOU - OK"
done
